async function send(){
  const v=id=>$("#"+id).value.trim(),w=gw();
  const com=COM.length?v("com"):v("comT");
  const err=!size?t("e1"):!v("nom")?t("e2"):!/^0[567]\d{8}$/.test(v("tel").replace(/\s/g,""))?t("e3"):(mode===0&&!v("adr"))?t("e6"):!w?t("e4"):!com?t("e5"):!ORDER_EMAIL?t("nocfg"):"";
  $("#err").textContent=err;if(err)return;

  const zz=rate(w),z=zz?zz[mode]:0,
        no="WDL-"+Date.now().toString(36).slice(-6).toUpperCase();

  const body={
    _subject:"Nouvelle commande "+no+" — WADOUD DE LUXE",
    _template:"table",_captcha:"false",
    "N° commande":no,"Produit":P.name,"Taille":size,"Quantité":qty,
    "Sous-total":P.price*qty+" DA",
    "Livraison":(mode?"Bureau (stop desk)":"À domicile")+" — "+(zz?z+" DA":"tarif à confirmer"),
    "TOTAL À PAYER":zz?(P.price*qty+z)+" DA":(P.price*qty)+" DA + livraison à confirmer",
    "Nom":v("nom"),"Téléphone":v("tel"),"Wilaya":w.c+" - "+w.fr,
    "Commune":com,"Adresse":v("adr")||"(bureau)","Langue du client":lg
  };

  const b=$("#go");
  b.disabled=true;
  b.classList.add("sending");
  $("#err").textContent=t("sending");

  try{
    const r=await fetch("https://formsubmit.co/ajax/"+ORDER_EMAIL,{
      method:"POST",
      headers:{
        "Content-Type":"application/json",
        "Accept":"application/json"
      },
      body:JSON.stringify(body),
      keepalive:true
    });
    const j=await r.json();
    if(j.success!==true&&j.success!=="true")throw new Error("send");
    $("#no").textContent=no;
    $("#form").hidden=true;
    $("#ok").hidden=false;
    $("#panel").scrollTop=0;
    $("#err").textContent="";
  }catch(e){
    $("#err").textContent=t("fail");
  }finally{
    b.disabled=false;
    b.classList.remove("sending");
  }
}
